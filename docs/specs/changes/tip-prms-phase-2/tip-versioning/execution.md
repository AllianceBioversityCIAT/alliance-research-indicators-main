#### JD-12 — TIP backout recipe (after a full TIP sync has run)

Use only when the migration must be reverted **after** step 2 of the rollout (design §11). Requires the TIP-scoped snapshot taken **after** the migration and **before** the first full sync (design §11 step 2).

```sql
-- 1. Snapshot, after the migration and before the sync.
DROP TABLE IF EXISTS t07_presync_tip;
DROP TABLE IF EXISTS t07_presync_tip_links;

CREATE TABLE t07_presync_tip AS
SELECT * FROM results WHERE platform_code = 'TIP';

CREATE TABLE t07_presync_tip_links AS
SELECT lr.* FROM link_results lr
WHERE EXISTS (
  SELECT 1 FROM t07_presync_tip t
  WHERE t.result_id = lr.result_id OR t.result_id = lr.other_result_id
);

-- 2. TIP result ids created after the snapshot.
--    Then, one at a time: SELECT full_delete_result_version(<result_id>);
SELECT r.result_id
FROM results r
LEFT JOIN t07_presync_tip p ON p.result_id = r.result_id
WHERE r.platform_code = 'TIP' AND p.result_id IS NULL;

-- 3. Cleanup. No DELETE of link_results that are merely absent from the snapshot.
UPDATE results r
INNER JOIN t07_presync_tip p ON p.result_id = r.result_id
SET r.`created_at` = p.`created_at`, r.`created_by` = p.`created_by`, r.`updated_at` = p.`updated_at`, r.`updated_by` = p.`updated_by`, r.`is_active` = p.`is_active`, r.`result_official_code` = p.`result_official_code`, r.`version_id` = p.`version_id`, r.`title` = p.`title`, r.`description` = p.`description`, r.`indicator_id` = p.`indicator_id`, r.`geo_scope_id` = p.`geo_scope_id`, r.`result_status_id` = p.`result_status_id`, r.`deleted_at` = p.`deleted_at`, r.`report_year_id` = p.`report_year_id`, r.`tip_id` = p.`tip_id`, r.`is_snapshot` = p.`is_snapshot`, r.`is_ai` = p.`is_ai`, r.`comment_geo_scope` = p.`comment_geo_scope`, r.`platform_code` = p.`platform_code`, r.`is_partner_not_applicable` = p.`is_partner_not_applicable`, r.`external_link` = p.`external_link`, r.`public_link` = p.`public_link`, r.`is_synced_to_prms` = p.`is_synced_to_prms`, r.`prms_result_code` = p.`prms_result_code`, r.`prms_phase_id` = p.`prms_phase_id`;

INSERT INTO results
SELECT p.* FROM t07_presync_tip p
LEFT JOIN results r ON r.result_id = p.result_id
WHERE r.result_id IS NULL;

UPDATE link_results lr
INNER JOIN t07_presync_tip_links p ON p.link_result_id = lr.link_result_id
SET lr.`created_at` = p.`created_at`, lr.`created_by` = p.`created_by`, lr.`updated_at` = p.`updated_at`, lr.`updated_by` = p.`updated_by`, lr.`is_active` = p.`is_active`, lr.`result_id` = p.`result_id`, lr.`other_result_id` = p.`other_result_id`, lr.`link_result_role_id` = p.`link_result_role_id`, lr.`deleted_at` = p.`deleted_at`;

INSERT INTO link_results
SELECT p.* FROM t07_presync_tip_links p
LEFT JOIN link_results lr ON lr.link_result_id = p.link_result_id
WHERE lr.link_result_id IS NULL;
```
